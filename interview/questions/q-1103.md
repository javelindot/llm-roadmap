# 使用numpy手撕二分类交叉熵，包括train和predict的函数

## 答案

**用线性 logits、稳定的 BCE 和梯度下降实现逻辑回归。** 对 z=Xw+b，p=sigmoid(z)，loss=mean(logaddexp(0,z)-y*z)，梯度为 X^T(p-y)/m 与 mean(p-y)。
```python
import numpy as np

class BinaryLogistic:
    def __init__(self):
        """初始化未训练状态。"""
        self.w = None
        self.b = 0.0

    def train(self, X, y, lr=0.1, epochs=1000):
        """在非空二维数据上训练，返回每步更新前的交叉熵。"""
        X = np.asarray(X, dtype=float)
        y = np.asarray(y, dtype=float)
        if X.ndim != 2 or len(X) == 0 or y.shape != (len(X),):
            raise ValueError("invalid shapes")
        if not np.isfinite(X).all() or not np.isin(y, [0, 1]).all():
            raise ValueError("invalid data")
        if lr <= 0 or not np.isfinite(lr) or epochs <= 0:
            raise ValueError("invalid training configuration")
        self.w = np.zeros(X.shape[1])
        self.b = 0.0
        losses = []
        for _ in range(epochs):
            z = X @ self.w + self.b
            p = np.exp(-np.logaddexp(0, -z))
            losses.append(float(np.mean(np.logaddexp(0, z) - y * z)))
            error = p - y
            self.w -= lr * (X.T @ error) / len(X)
            self.b -= lr * error.mean()
        return losses

    def predict_proba(self, X):
        """使用稳定 sigmoid 计算正类概率，要求模型已训练。"""
        X = np.asarray(X, dtype=float)
        if self.w is None or X.ndim != 2 or X.shape[1] != len(self.w):
            raise ValueError("untrained model or invalid shape")
        if not np.isfinite(X).all():
            raise ValueError("invalid data")
        return np.exp(-np.logaddexp(0, -(X @ self.w + self.b)))

    def predict(self, X, threshold=0.5):
        """按指定概率阈值输出二分类标签。"""
        if not 0 <= threshold <= 1:
            raise ValueError("invalid threshold")
        return (self.predict_proba(X) >= threshold).astype(int)
```
训练每轮 O(md)，参数空间 O(d)，批量中间结果 O(m)。特征宜标准化，学习率过大会发散；用可分样本检查 loss 下降、预测正确和极端 logits 数值稳定。
